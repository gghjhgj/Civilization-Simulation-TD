# ============================================================
# WSPÓLNE FLAGI KOMPILATORA
# ============================================================

if(MSVC)

    target_compile_options(
        simulation_core
        PRIVATE
        /W3
    )

    target_compile_options(
        app
        PRIVATE
        /W3
    )

else()

    target_compile_options(
        simulation_core
        PRIVATE
        -Wall
        -Wextra
        -funroll-loops

        $<$<BOOL:${ENABLE_NATIVE_ARCH}>:-march=native>
        $<$<BOOL:${ENABLE_AVX2}>:-mavx2>
        $<$<BOOL:${ENABLE_FAST_MATH}>:-ffast-math>

        $<$<AND:$<NOT:$<BOOL:${ENABLE_CRASH_DEBUG}>>,$<CONFIG:Release>>:-Ofast>
        $<$<AND:$<BOOL:${ENABLE_CRASH_DEBUG}>,$<CONFIG:Release>>:-O1>

        $<$<AND:$<CONFIG:Debug>,$<NOT:$<BOOL:${ENABLE_CRASH_DEBUG}>>>:-g;-O0>

        $<$<BOOL:${ENABLE_CRASH_DEBUG}>:-g;-O1;-fno-omit-frame-pointer>
    )

    target_compile_options(
        app
        PRIVATE
        -Wall
        -Wextra

        $<$<BOOL:${ENABLE_NATIVE_ARCH}>:-march=native>
        $<$<BOOL:${ENABLE_AVX2}>:-mavx2>
        $<$<BOOL:${ENABLE_FAST_MATH}>:-ffast-math>

        $<$<AND:$<NOT:$<BOOL:${ENABLE_CRASH_DEBUG}>>,$<CONFIG:Release>>:-Ofast>
        $<$<AND:$<BOOL:${ENABLE_CRASH_DEBUG}>,$<CONFIG:Release>>:-O1>

        $<$<AND:$<CONFIG:Debug>,$<NOT:$<BOOL:${ENABLE_CRASH_DEBUG}>>>:-g;-O0>

        $<$<BOOL:${ENABLE_CRASH_DEBUG}>:-g;-O1;-fno-omit-frame-pointer>
    )

endif()

    ENABLE_GCC_PGO_GENERATE

        FATAL_ERROR
        "[CMake] ENABLE_GCC_PGO_GENERATE i ENABLE_GCC_PGO_USE nie moga byc jednoczesnie ON."
    )

if(
    ENABLE_GCC_PGO_GENERATE
    OR ENABLE_GCC_PGO_USE
)

    if(NOT CMAKE_CXX_COMPILER_ID STREQUAL "GNU")

        message(
            FATAL_ERROR
            "[CMake] GCC PGO wymaga kompilatora GCC."
        )

    endif()

    if(ENABLE_GCC_PGO_GENERATE)

        message(
            STATUS
            "[CMake] GCC PGO: GENERATE"
        )

        target_compile_options(
            simulation_core
            PRIVATE
            "-fprofile-generate=${GCC_PROFILE_DIR}"
        )

        target_compile_options(
            app
            PRIVATE
            "-fprofile-generate=${GCC_PROFILE_DIR}"
        )

        target_link_options(
            app
            PRIVATE
            "-fprofile-generate=${GCC_PROFILE_DIR}"
        )

    elseif(ENABLE_GCC_PGO_USE)

        message(
            STATUS
            "[CMake] GCC PGO: USE"
        )

        target_compile_options(
            simulation_core
            PRIVATE
            "-fprofile-use=${GCC_PROFILE_DIR}"
            "-fprofile-correction"
        )

        target_compile_options(
            app
            PRIVATE
            "-fprofile-use=${GCC_PROFILE_DIR}"
            "-fprofile-correction"
        )

        target_link_options(
            app
            PRIVATE
            "-fprofile-use=${GCC_PROFILE_DIR}"
        )

    endif()

endif()

set(
    ASSEMBLY_DIR
    "${CMAKE_CURRENT_SOURCE_DIR}/assembly"
)

add_custom_target(
    assembler

    COMMAND
        ${CMAKE_COMMAND}
        -E
        remove_directory
        "${ASSEMBLY_DIR}"

    COMMAND
        ${CMAKE_COMMAND}
        -E
        make_directory
        "${ASSEMBLY_DIR}"

    COMMENT
        "[CMake] Generowanie asemblera..."
)

foreach(SOURCE_FILE IN LISTS PROJECT_SOURCES)

    file(
        RELATIVE_PATH
        SOURCE_RELATIVE
        "${CMAKE_CURRENT_SOURCE_DIR}"
        "${SOURCE_FILE}"
    )

    get_filename_component(
        SOURCE_DIR
        "${SOURCE_RELATIVE}"
        DIRECTORY
    )

    get_filename_component(
        SOURCE_NAME
        "${SOURCE_RELATIVE}"
        NAME_WE
    )

    set(
        ASSEMBLY_OUTPUT_DIR
        "${ASSEMBLY_DIR}/${SOURCE_DIR}"
    )

    set(
        ASSEMBLY_OUTPUT
        "${ASSEMBLY_OUTPUT_DIR}/${SOURCE_NAME}.s"
    )

    add_custom_command(
        TARGET assembler

        COMMAND
            ${CMAKE_COMMAND}
            -E
            make_directory
            "${ASSEMBLY_OUTPUT_DIR}"

        COMMAND
            ${CMAKE_CXX_COMPILER}
            $<$<BOOL:${ENABLE_NATIVE_ARCH}>:-march=native>
            $<$<BOOL:${ENABLE_AVX2}>:-mavx2>
            $<$<BOOL:${ENABLE_FAST_MATH}>:-ffast-math>
            $<$<CONFIG:Release>:-Ofast>
            $<$<CONFIG:Debug>:-O0>
            -funroll-loops
            -S
            -fverbose-asm
            -I"${CMAKE_CURRENT_SOURCE_DIR}"
            "${SOURCE_FILE}"
            -o
            "${ASSEMBLY_OUTPUT}"

        VERBATIM
    )

endforeach()
